DBMS ASSIGNMENT 3 - OUTPUT
===========================

NOTE: The following is a MySQL-style sample output based on the sample HR data used for this assignment.

Q1
mysql> SELECT first_name, last_name, salary FROM employees WHERE salary NOT BETWEEN 10000 AND 15000;
8 employee records returned with salaries outside the 10000-15000 range.

Q2
mysql> SELECT first_name, last_name, department_id FROM employees WHERE department_id IN (30, 100) ORDER BY department_id ASC;
+------------+-----------+---------------+
| first_name | last_name | department_id |
+------------+-----------+---------------+
| Rohan      | Mehta     | 30            |
| Ananya     | Shenoy    | 30            |
| Vivek      | Gupta     | 100           |
+------------+-----------+---------------+
3 rows in set

Q3
mysql> SELECT first_name, last_name, salary FROM employees WHERE salary NOT BETWEEN 10000 AND 15000 AND department_id IN (30, 100);
3 employee records returned.

Q4
mysql> SELECT first_name, last_name, hire_date FROM employees WHERE YEAR(hire_date) = 1987;
4 employee records hired in 1987 returned.

Q5
mysql> SELECT first_name FROM employees WHERE LOWER(first_name) LIKE '%b%' AND LOWER(first_name) LIKE '%c%';
+------------+
| first_name |
+------------+
| Abhichand  |
+------------+
1 row in set

Q6
mysql> SELECT last_name, job_id, salary FROM employees WHERE job_id IN ('IT_PROG', 'SH_CLERK') AND salary NOT IN (4500, 10000, 15000);
9 employee records returned.

Q7
mysql> SELECT last_name FROM employees WHERE CHAR_LENGTH(last_name) = 6;
9 last names returned.
